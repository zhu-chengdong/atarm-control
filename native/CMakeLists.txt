cmake_minimum_required(VERSION 3.10)
project(atarm_serial_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ATARM_ENABLE_PHYSICAL_UART "Allow already-configured physical UART descriptors (not tested)" OFF)
link_directories(${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib)
add_library(a1_serial_adapter SHARED a1_serial.cpp)
target_include_directories(a1_serial_adapter PRIVATE vendor/include)
target_compile_options(a1_serial_adapter PRIVATE -Wall -Wextra)
target_link_libraries(a1_serial_adapter PRIVATE UnitreeMotorSDK_Linux64 pthread)
if(ATARM_ENABLE_PHYSICAL_UART)
  target_compile_definitions(a1_serial_adapter PRIVATE ATARM_ENABLE_PHYSICAL_UART=1)
endif()
add_executable(sdk_codec_probe sdk_codec_probe.cpp)
target_include_directories(sdk_codec_probe PRIVATE vendor/include)
target_link_libraries(sdk_codec_probe PRIVATE UnitreeMotorSDK_Linux64)
set_target_properties(a1_serial_adapter sdk_codec_probe PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../vendor/lib")
